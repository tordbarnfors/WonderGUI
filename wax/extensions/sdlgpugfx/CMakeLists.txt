#------------------------------------------------------------------------------
# Shaders
#
# compile_shaders.py compiles the HLSL shaders in ../hlsl to SPIR-V, DXIL and
# MSL in shaders/, which are checked in. Here those SDL_GPU can use on the
# platform are turned into arrays in a source file of the build directory, so
# that building needs no shader compiler, see wax_sdlgpushaders.h: SPIR-V for
# Vulkan everywhere but on Apple's platforms, which only have Metal, and DXIL
# for D3D12 on Windows.
#------------------------------------------------------------------------------

set(SHADER_FORMATS)

if(APPLE)
	list(APPEND SHADER_FORMATS msl)
else()
	list(APPEND SHADER_FORMATS spv)

	if(WIN32)
		list(APPEND SHADER_FORMATS dxil)
	endif()
endif()

set(SHADER_SOURCE "// Generated by CMake from the files in wax/extensions/sdlgpugfx/shaders.\n\n#include <wax_sdlgpushaders.h>\n\nnamespace wax\n{\nnamespace SdlGpuShaders\n{\n")
set(SHADER_FILES)

foreach(FORMAT spv dxil msl)

	if(FORMAT STREQUAL "spv")
		set(TABLE spirv)
	else()
		set(TABLE ${FORMAT})
	endif()

	set(ENTRIES "")

	if(FORMAT IN_LIST SHADER_FORMATS)

		file(GLOB FILES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/shaders/*.${FORMAT})
		list(SORT FILES)
		list(APPEND SHADER_FILES ${FILES})

		foreach(FILE ${FILES})
			get_filename_component(SHADER_NAME ${FILE} NAME_WE)
			file(READ ${FILE} HEX HEX)

			string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," BYTES "${HEX}")

			# SPIR-V is read as 32-bit words.

			string(APPEND SHADER_SOURCE "\talignas(4) static const uint8_t s_${SHADER_NAME}_${FORMAT}[] = { ${BYTES} };\n\n")
			string(APPEND ENTRIES "\t\t{ \"${SHADER_NAME}\", s_${SHADER_NAME}_${FORMAT}, sizeof(s_${SHADER_NAME}_${FORMAT}) },\n")
		endforeach()
	endif()

	if(ENTRIES STREQUAL "")
		string(APPEND SHADER_SOURCE "\tconst Code ${TABLE}[] = { { nullptr, nullptr, 0 } };\n\tconst int nb${TABLE}Count = 0;\n\n")
	else()
		string(APPEND SHADER_SOURCE "\tconst Code ${TABLE}[] =\n\t{\n${ENTRIES}\t};\n\n\tconst int nb${TABLE}Count = int(sizeof(${TABLE}) / sizeof(${TABLE}[0]));\n\n")
	endif()

endforeach()

# The counts by the names wax_sdlgpushaders.h has for them.

string(APPEND SHADER_SOURCE "\tconst int nbSpirv = nbspirvCount;\n\tconst int nbDxil = nbdxilCount;\n\tconst int nbMsl = nbmslCount;\n}\n}\n")

set(SHADER_CPP ${CMAKE_CURRENT_BINARY_DIR}/wax_sdlgpushadercode.cpp)

# Only written when it changes, so that it isn't compiled again for nothing.

file(CONFIGURE OUTPUT ${SHADER_CPP} CONTENT "${SHADER_SOURCE}" @ONLY)

set(SHADERS
	compile_shaders.py
	${SHADER_FILES}
)

source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR} FILES ${SOURCES} ${SHADERS})
set_source_files_properties(${SHADERS} PROPERTIES HEADER_FILE_ONLY TRUE)

add_library( waxsdlgpugfx STATIC ${SOURCES} ${SHADER_CPP} ${SHADERS} )

target_include_directories(waxsdlgpugfx PRIVATE
	${WAX_INCDIRS}
	${INCDIRS}
)

# Public, so whatever links waxsdlgpugfx gets SDL too.

target_link_libraries(waxsdlgpugfx PUBLIC wax ${SDL3_LIBS})

list(TRANSFORM SOURCES PREPEND ${CMAKE_CURRENT_LIST_DIR}/ )
set(WAXSDLGPUGFX_SOURCES ${SOURCES} PARENT_SCOPE)
set(WAXSDLGPUGFX_INCDIRS ${INCDIRS} PARENT_SCOPE)
