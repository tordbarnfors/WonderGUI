set(SOURCES

	wax_vulkanbackend.cpp
	wax_vulkanbackend.h
	wax_vulkancontext.cpp
	wax_vulkancontext.h
	wax_vulkanshaders.h
	wax_vulkansurface.cpp

	wax_vulkanedgemap.cpp

set(INCDIRS
	${CMAKE_CURRENT_LIST_DIR}
)

#------------------------------------------------------------------------------
# Vulkan
#
# Only looked for here, so nothing needs Vulkan unless this extension is built.
# We link the loader, which has all of Vulkan 1.3 but extensions, which
# VulkanBackend gets through vkGetDeviceProcAddr().
#------------------------------------------------------------------------------

find_package(Vulkan REQUIRED)

#------------------------------------------------------------------------------
# Shaders
#
# compile_shaders.py compiles the HLSL shaders in ../hlsl to SPIR-V in spirv/,
# which is checked in. Here they are turned into arrays in a source file of the
# build directory, so that building needs no shader compiler. See
# wax_vulkanshaders.h for their names.
#------------------------------------------------------------------------------

file(GLOB SPIRV_FILES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/spirv/*.spv)
list(SORT SPIRV_FILES)

set(SHADER_SOURCE "// Generated by CMake from the files in wax/extensions/vulkangfx/spirv.\n\n#include <wax_vulkanshaders.h>\n\nnamespace wax\n{\nnamespace VulkanShaders\n{\n")

foreach(SPIRV_FILE ${SPIRV_FILES})
	get_filename_component(SHADER_NAME ${SPIRV_FILE} NAME_WE)
	file(READ ${SPIRV_FILE} HEX HEX)

	# SPIR-V is a sequence of little endian 32-bit words.

	string(REGEX REPLACE "([0-9a-f][0-9a-f])([0-9a-f][0-9a-f])([0-9a-f][0-9a-f])([0-9a-f][0-9a-f])" "0x\\4\\3\\2\\1," WORDS "${HEX}")

	string(APPEND SHADER_SOURCE "\tstatic const uint32_t s_${SHADER_NAME}[] = { ${WORDS} };\n")
	string(APPEND SHADER_SOURCE "\tconst Code ${SHADER_NAME} = { s_${SHADER_NAME}, sizeof(s_${SHADER_NAME}) };\n\n")
endforeach()

string(APPEND SHADER_SOURCE "}\n}\n")

set(SHADER_CPP ${CMAKE_CURRENT_BINARY_DIR}/wax_vulkanshaders.cpp)

# Only written when it changes, so that it isn't compiled again for nothing.

file(CONFIGURE OUTPUT ${SHADER_CPP} CONTENT "${SHADER_SOURCE}" @ONLY)

set(SHADERS
	compile_shaders.py
	${SPIRV_FILES}
)

source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR} FILES ${SOURCES} ${SHADERS})
set_source_files_properties(${SHADERS} PROPERTIES HEADER_FILE_ONLY TRUE)

add_library( waxvulkangfx STATIC ${SOURCES} ${SHADER_CPP} ${SHADERS} )

target_include_directories(waxvulkangfx PRIVATE
	${WAX_INCDIRS}
	${INCDIRS}
)

# Public, so whatever links waxvulkangfx gets the Vulkan headers and loader too.

target_link_libraries(waxvulkangfx PUBLIC wax Vulkan::Vulkan)

list(TRANSFORM SOURCES PREPEND ${CMAKE_CURRENT_LIST_DIR}/ )
set(WAXVULKANGFX_SOURCES ${SOURCES} PARENT_SCOPE)
set(WAXVULKANGFX_INCDIRS ${INCDIRS} PARENT_SCOPE)
